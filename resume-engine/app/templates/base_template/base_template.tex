% Resume Bullet

    \item \small{

    }

% Experience / Education Heading
\newcommand{\resumeSubheading}[4]{
    \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]
    {l@{\extracolsep{\fill}}r}

        \textbf{#1} & #2 \\

        \textit{\small #3} &
        \textit{\small #4} \\

    \end{tabular*}

}

% Project Heading
\newcommand{\resumeProjectHeading}[2]{
    \item

}
